// Function to sort parcels by weight
void sortByWeight(struct Parcel p[], int n) {

struct Parcel temp;
int swaps = 0;
int bytes = 0;

for (int i = 0; i < n - 1; i++) {

for (int j = 0; j < n - i - 1; j++) {

if (p[j].weight > p[j + 1].weight) {

temp = p[j];
p[j] = p[j + 1];
p[j + 1] = temp;

swaps++;
bytes += 3 * sizeof(struct Parcel);
}
}
}

printf("\nParcels sorted by weight!\n");
printf("Total swaps: %d\n", swaps);
printf("Bytes copied: %d\n", bytes);
}

struct Date {
int day, month, year;
};

struct Time {
int hour, min;
};

enum ItemType { PACKAGED, COOKED, COMBO };

struct Item {
int id;
char name[30];
float price;
int stock;
int sold;
enum ItemType type;
union {
struct Date expiry; /* For PACKAGED */
struct Time prep_time; /* For COOKED */
int combo_ids[3]; /* For COMBO (up to 3 item IDs) */
} detail;
};

/* Helper: Convert Date to total days for simple comparison */
int toDays(struct Date d) {
return d.year * 365 + d.month * 30 + d.day;
}

/* Record a sale safely using structure pointer */
void recordSale(struct Item *item, int qty) {
static int bill_no = 100; /* Static counter persists across calls */
bill_no++;

if (qty <= 0) {
printf("[Bill #%d] Invalid quantity for %s.\n", bill_no, item->name);
return;
}
if (qty > item->stock) {
printf("[Bill #%d] FAILED: Requested %d of '%s', but only %d in stock.\n",
bill_no, qty, item->name, item->stock);
return;
}

item->stock -= qty;
item->sold += qty;
printf("[Bill #%d] SOLD: %d x %s = Rs. %.2f (Remaining stock: %d)\n",
bill_no, qty, item->name, qty * item->price, item->stock);
}

/* Warn about packaged items expiring within 2 days */
void checkExpiry(const struct Item *menu, int n, struct Date today) {
printf("\n--- Expiry Warnings (Within 2 Days of %02d/%02d/%d) ---\n",
today.day, today.month, today.year);
int found = 0;
for (int i = 0; i < n; i++) {
if (menu[i].type == PACKAGED) {
int diff = toDays(menu[i].detail.expiry) - toDays(today);
if (diff >= 0 && diff <= 2) {
printf("WARNING: '%s' expires in %d day(s) (%02d/%02d/%d)!\n",
menu[i].name, diff,
menu[i].detail.expiry.day,
menu[i].detail.expiry.month,
menu[i].detail.expiry.year);
found = 1;
}
}
}
if (!found) printf("No items expiring soon.\n");
}

/* Sort items by revenue (descending) and print summary */
void printSortedSummary(struct Item *menu, int n) {
for (int i = 0; i < n - 1; i++) {
for (int j = 0; j < n - i - 1; j++) {
float rev1 = menu[j].sold * menu[j].price;
float rev2 = menu[j + 1].sold * menu[j + 1].price;
if (rev1 < rev2) {
struct Item temp = menu[j];
menu[j] = menu[j + 1];
menu[j + 1] = temp;
}
}
}

printf("\n--- Day's Bill Summary (Sorted by Revenue) ---\n");
printf("%-18s %-6s %-8s %-10s\n", "Item", "Sold", "Stock", "Revenue");
float total_rev = 0;
for (int i = 0; i < n; i++) {
float rev = menu[i].sold * menu[i].price;
total_rev += rev;
printf("%-18s %-6d %-8d Rs. %.2f\n",
menu[i].name, menu[i].sold, menu[i].stock, rev);
}
printf("Total Canteen Revenue: Rs. %.2f\n", total_rev);
}

int main(void) {
struct Date today = {14, 11, 2026};

struct Item menu[4] = {
{1, "Mango Juice", 30.0f, 10, 0, PACKAGED, {.expiry = {15, 11, 2026}}},
{2, "Veg Biryani", 80.0f, 15, 0, COOKED, {.prep_time = {12, 30}}},
{3, "Lunch Combo", 100.0f, 5, 0, COMBO, {.combo_ids = {1, 2, 0}}},
{4, "Biscuit Pack", 20.0f, 20, 0, PACKAGED, {.expiry = {25, 11, 2026}}}

printf("--- Recording Sales ---\n");
recordSale(&menu[0], 3); /* Normal sale */

}
